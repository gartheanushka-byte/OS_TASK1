<script>
  function populate(target) {
    const area = document.getElementById(target);
    for (let i = 0; i < 25; i++) {
      const cell = document.createElement("span");
      cell.textContent = 1 + Math.floor(Math.random() * 9);
      area.appendChild(cell);
    }
  }

  ["left", "right", "answer"].forEach(populate);

  let cycle = 0;
  setInterval(() => {
    cycle += 1;
    const row = cycle % 5;
    const col = (cycle * 2) % 5;

    document.getElementById("worker").textContent =
      `Worker contribution: A[${row}][k] × B[k][${col}]`;

    document.getElementById("status").textContent =
      `Processing partial contribution ${cycle}; synchronized result updates are represented by the worker activity.`;
  }, 700);
</script>
